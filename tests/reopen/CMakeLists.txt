# scene.reopen_fidelity — the save/close/reopen round trip (2026-09-04).
#
# Drives the real binary with --script, like the scripting.e2e.* suites, and for
# the same reason: the defect it gates only exists when the writer, the reader,
# the asset catalog, the store and the renderer are all the live ones. HOME is
# redirected to a scratch dir so the run gets a fresh library and can never
# touch the user's. It needs a reachable display (it probes pixels) — the same
# contract every pixel e2e suite here has.
#
# REBUILT TO ITS LAW (TEST-CUT-1; CLAUDE.md Testing: every row at most 60 s): the script's three arms are independent —
# each its own project, its own reference taken fresh in the same process — so the one row (74 s, 2,774 frames, eight
# reopens) runs as THREE, each arm whole: every cycle kept (three main, two high-sky, three scaled), every settle the
# renderer's own predicate (GI at rest, no probe owing a capture), every bar unchanged. REOPEN_PART names the arm
# (a generated copy of the script per row; the high-sky arm takes a fresh scene of its own, which is what the main
# arm's three cycles proved its scene to be). Each row's home is wiped at the start of every run (PROJECT-NAMES-1:
# a project name is unique in a library). Measured once each (2026-10-20, solo, the card lock held): main 47.4 s / 1,510
# frames, high_sky 39.5 s / 1,115, scaled 30.2 s / 506. TIMEOUT 6 x the solo run, rounded up.
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/scripts/reopen_fidelity.js _reopen_js)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/scripts/reopen_fidelity.js)
set(_reopen_timeout_main 60)
set(_reopen_timeout_high_sky 60)
set(_reopen_timeout_scaled 60)
foreach(_part main high_sky scaled)
    if(_part STREQUAL "main")
        set(_row scene.reopen_fidelity)
    else()
        set(_row scene.reopen_fidelity.${_part})
    endif()
    set(_home ${CMAKE_CURRENT_BINARY_DIR}/reopen-home-${_part})
    file(MAKE_DIRECTORY ${_home}/run)
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/reopen_fidelity_${_part}.js
         "var REOPEN_PART = \"${_part}\";   // generated by tests/reopen/CMakeLists.txt\n${_reopen_js}")
    jah_gpu_row(${_row} SHIP CLASS app TIER document WARM_HOME "${_home}"
             COMMAND $<TARGET_FILE:${CMAKE_PROJECT_NAME}> --script ${CMAKE_CURRENT_BINARY_DIR}/reopen_fidelity_${_part}.js
             WORKING_DIRECTORY ${_home}/run)
    set_tests_properties(${_row} PROPERTIES
        ENVIRONMENT "HOME=${_home};JAHSHAKA_DATA_ROOT=${_home}/.local/share/Jahshaka"
        LABELS "scripting"
        TIMEOUT ${_reopen_timeout_${_part}})
    jah_fresh_home_fixture(${_row}.fresh_home sceneReopenFidelityFreshHome_${_part} "${_home}" WARM LABELS scripting)
    set_property(TEST ${_row} APPEND PROPERTY FIXTURES_REQUIRED sceneReopenFidelityFreshHome_${_part})
endforeach()

# THE FRAMES EACH ROW RENDERS (TESTING-SPEED-B2; SPECS/TESTING_SPEED_SPEC.md §2 A8): the test property
# JAHSHAKA_TEST_FRAMES, MEASURED — every engine frame (Engine::renderOneFrame) of every process the row starts,
# the boot's startup-gate frames included, counted on this tree's Debug build on 2026-10-12 (the count and its
# method: spikes/testing-speed-b2/frames_pass1.tsv). A pool's is the sum over its arms. A headless row renders
# the one empty frame a headless engine walks.
if(TEST scene.reopen_fidelity)
    set_property(TEST scene.reopen_fidelity PROPERTY JAHSHAKA_TEST_FRAMES 1510)
    set_property(TEST scene.reopen_fidelity.high_sky PROPERTY JAHSHAKA_TEST_FRAMES 1115)
    set_property(TEST scene.reopen_fidelity.scaled PROPERTY JAHSHAKA_TEST_FRAMES 506)
endif()
